-- INVOKER guards retain the invoking outer finalizer's effective role.
-- Marking these SECURITY DEFINER as that same writer would make the role check tautological.
CREATE FUNCTION public.source22_header_insert_guard() RETURNS trigger
LANGUAGE plpgsql SECURITY INVOKER SET search_path=pg_catalog SET row_security=on AS $$
BEGIN
  IF NEW.source_effect_protocol IS NOT NULL THEN
    IF current_user <> 'console_owner_commit_writer' THEN
      RAISE EXCEPTION 'source receipt writer refused' USING ERRCODE='42501';
    END IF;
    IF NEW.org_id IS DISTINCT FROM NULLIF(pg_catalog.current_setting('app.current_org',true),'')::uuid THEN
      RAISE EXCEPTION 'source receipt scope refused' USING ERRCODE='42501';
    END IF;
  END IF;
  RETURN NEW;
END;
$$;
CREATE TRIGGER source22_header_insert_guard BEFORE INSERT ON public.ont_action_command_receipts
  FOR EACH ROW EXECUTE FUNCTION public.source22_header_insert_guard();

CREATE FUNCTION public.source22_member_insert_guard() RETURNS trigger
LANGUAGE plpgsql SECURITY INVOKER SET search_path=pg_catalog SET row_security=on AS $$
DECLARE
  v_protocol text;
  v_count smallint;
  v_xmin xid;
BEGIN
  IF current_user <> 'console_owner_commit_writer' THEN
    RAISE EXCEPTION 'source member writer refused' USING ERRCODE='42501';
  END IF;
  IF NEW.org_id IS DISTINCT FROM NULLIF(pg_catalog.current_setting('app.current_org',true),'')::uuid THEN
    RAISE EXCEPTION 'source member scope refused' USING ERRCODE='42501';
  END IF;
  SELECT r.source_effect_protocol,r.source_effect_count,r.xmin
    INTO v_protocol,v_count,v_xmin FROM public.ont_action_command_receipts r
    WHERE r.org_id=NEW.org_id AND r.command_id=NEW.command_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'source manifest header unavailable' USING ERRCODE='23514';
  END IF;
  IF v_protocol IS DISTINCT FROM 'SOURCE_EFFECTS21'
     OR v_count IS NULL OR v_count NOT BETWEEN 1 AND 1024
     OR v_xmin IS DISTINCT FROM pg_catalog.pg_current_xact_id()::xid
     OR NEW.ordinal IS NULL OR NEW.ordinal<0 OR NEW.ordinal>=v_count THEN
    RAISE EXCEPTION 'source manifest append refused' USING ERRCODE='23514';
  END IF;
  RETURN NEW;
END;
$$;
CREATE TRIGGER source22_member_insert_guard BEFORE INSERT ON public.ont_action_source_effects
  FOR EACH ROW EXECUTE FUNCTION public.source22_member_insert_guard();

-- Deferred execution may occur after the outer SECURITY DEFINER returned to LOGIN.
-- This fixed read-only validation function therefore runs as the confined writer.
CREATE FUNCTION public.source22_validate_manifest() RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog SET row_security=on AS $$
DECLARE
  v_count bigint;
  v_min smallint;
  v_max smallint;
  v_order_ok boolean;
  v_members bytea;
  v_header bytea;
  v_actual bytea;
BEGIN
  IF NEW.source_effect_protocol IS NULL THEN
    RETURN NULL;
  END IF;
  IF NEW.org_id IS DISTINCT FROM NULLIF(pg_catalog.current_setting('app.current_org',true),'')::uuid THEN
    RAISE EXCEPTION 'source manifest scope refused' USING ERRCODE='42501';
  END IF;
  IF NEW.source_effect_protocol IS DISTINCT FROM 'SOURCE_EFFECTS21'
     OR NEW.source_effect_count IS NULL OR NEW.source_effect_count NOT BETWEEN 1 AND 1024
     OR NEW.source_effect_digest IS NULL OR pg_catalog.octet_length(NEW.source_effect_digest)<>32 THEN
    RAISE EXCEPTION 'source manifest header invalid' USING ERRCODE='23514';
  END IF;
  -- Indexed(org,command,ordinal) selection; immediate guard bounds cardinality<=1024.
  -- Fixed115byte members, ordered aggregate rather than repeated growing bytea concatenation.
  WITH bounded AS MATERIALIZED (
    SELECT e.* FROM public.ont_action_source_effects e
    WHERE e.org_id=NEW.org_id AND e.command_id=NEW.command_id ORDER BY e.ordinal LIMIT 1025
  ), ordered AS (
    SELECT b.*,lag(b.family_code) OVER w AS pf,lag(b.root_id) OVER w AS pr,
      lag(b.revision) OVER w AS pv,lag(b.effect_kind) OVER w AS pk
    FROM bounded b WINDOW w AS(ORDER BY b.ordinal)
  ), framed AS (
    SELECT o.ordinal,
      (o.pf IS NULL OR ROW(o.family_code,o.root_id,o.revision,o.effect_kind)>ROW(o.pf,o.pr,o.pv,o.pk)) AS in_order,
      pg_catalog.set_byte(pg_catalog.decode('00','hex'),0,o.family_code::integer)
      ||pg_catalog.set_byte(pg_catalog.decode('00','hex'),0,o.effect_kind::integer)
      ||pg_catalog.uuid_send(o.root_id)||pg_catalog.int8send(o.revision)||o.custody_digest
      ||CASE WHEN o.effect_kind=2 THEN pg_catalog.decode('01','hex')||pg_catalog.uuid_send(o.successor_root_id)
        ||pg_catalog.int8send(o.successor_revision)||o.successor_custody_digest
        ELSE pg_catalog.decode(pg_catalog.repeat('00',57),'hex') END AS member_bytes
    FROM ordered o
  )
  SELECT count(*),min(ordinal),max(ordinal),bool_and(in_order),string_agg(member_bytes,''::bytea ORDER BY ordinal)
    INTO v_count,v_min,v_max,v_order_ok,v_members FROM framed;
  IF v_count<>NEW.source_effect_count OR v_min IS DISTINCT FROM 0::smallint
     OR v_max IS DISTINCT FROM (NEW.source_effect_count-1)::smallint OR v_order_ok IS NOT TRUE
     OR v_members IS NULL OR pg_catalog.octet_length(v_members)<>115*NEW.source_effect_count THEN
    RAISE EXCEPTION 'source manifest membership invalid' USING ERRCODE='23514';
  END IF;
  v_header:=pg_catalog.convert_to('console.source21.effects','UTF8')||pg_catalog.decode('00','hex')
    ||pg_catalog.uuid_send(NEW.org_id)||pg_catalog.uuid_send(NEW.command_id)
    ||pg_catalog.int4send(NEW.source_effect_count::integer);
  v_actual:=pg_catalog.sha256(v_header||v_members);
  IF v_actual IS DISTINCT FROM NEW.source_effect_digest THEN
    RAISE EXCEPTION 'source manifest commitment invalid' USING ERRCODE='23514';
  END IF;
  RETURN NULL;
END;
$$;
ALTER FUNCTION public.source22_validate_manifest() OWNER TO console_owner_commit_writer;

  EXECUTE FUNCTION public.source22_validate_manifest();
