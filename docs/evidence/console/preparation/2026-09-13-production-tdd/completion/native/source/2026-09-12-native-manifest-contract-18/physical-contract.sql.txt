CREATE TABLE public.employee_contract_wage_corrections (
  org_id uuid NOT NULL, base_wage_id uuid NOT NULL, employee_id uuid NOT NULL,
  revision bigint NOT NULL CHECK(revision>0),
  predecessor_revision bigint NOT NULL CHECK(predecessor_revision>=0 AND predecessor_revision=revision-1),
  predecessor_fk_revision bigint GENERATED ALWAYS AS (NULLIF(predecessor_revision,0)) STORED,
  amount_won bigint NOT NULL CHECK(amount_won>0),
  wage_kind text NOT NULL CHECK(wage_kind IN ('MONTHLY','HOURLY')),
  monthly_standard_hours integer NOT NULL CHECK(monthly_standard_hours BETWEEN 1 AND 400),
  source_note text NOT NULL CHECK(char_length(source_note)<=500),
  reason text NOT NULL CHECK(char_length(reason) BETWEEN 1 AND 2000),
  evidence_refs jsonb NOT NULL CHECK(jsonb_typeof(evidence_refs)='array' AND jsonb_array_length(evidence_refs)<=32),
  actor_account_id uuid NOT NULL, command_id uuid NOT NULL,
  recorded_at_us bigint NOT NULL,
  payload_digest bytea NOT NULL CHECK(octet_length(payload_digest)=32),
  PRIMARY KEY(org_id,base_wage_id,revision),
  UNIQUE(org_id,base_wage_id,predecessor_revision),
  UNIQUE(org_id,command_id),
  FOREIGN KEY(org_id,base_wage_id,employee_id) REFERENCES public.employee_contract_wages(org_id,id,employee_id) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,base_wage_id,predecessor_fk_revision) REFERENCES public.employee_contract_wage_corrections(org_id,base_wage_id,revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,actor_account_id) REFERENCES public.company_actors(org_id,account_id) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,command_id) REFERENCES public.ont_action_command_receipts(org_id,command_id) ON DELETE RESTRICT DEFERRABLE INITIALLY DEFERRED
);

  profile_id uuid NOT NULL, profile_revision bigint NOT NULL CHECK(profile_revision>0),
  rule_bundle_id text NOT NULL, kernel_build_sha256 bytea NOT NULL CHECK(octet_length(kernel_build_sha256)=32),

  scope_id uuid NOT NULL, scope_revision bigint NOT NULL CHECK(scope_revision>0),
  membership_count integer NOT NULL CHECK(membership_count BETWEEN 0 AND 2000),
  membership_basis_digest bytea NOT NULL CHECK(octet_length(membership_basis_digest)=32),

CREATE TABLE public.payroll_input_units (
  org_id uuid NOT NULL, run_id uuid NOT NULL, input_revision bigint NOT NULL,
  employee_id uuid NOT NULL, line_id uuid NOT NULL,
  resolution text NOT NULL CHECK(resolution IN ('RESOLVED','BLOCKED')),
  gross_won bigint CHECK(gross_won>0), taxable_monthly_won bigint CHECK(taxable_monthly_won>=0),
  pension_standard_monthly_income_won bigint CHECK(pension_standard_monthly_income_won>=0),
  monthly_standard_hours integer CHECK(monthly_standard_hours BETWEEN 1 AND 400),
  income_tax_won bigint CHECK(income_tax_won>=0), local_tax_won bigint CHECK(local_tax_won>=0),
  tax_table_edition text,
  blockers jsonb NOT NULL CHECK(jsonb_typeof(blockers)='array' AND jsonb_array_length(blockers)<=32),
  semantic_digest bytea NOT NULL CHECK(octet_length(semantic_digest)=32),
  custody_digest bytea NOT NULL CHECK(octet_length(custody_digest)=32),
  PRIMARY KEY(org_id,run_id,input_revision,employee_id),
  UNIQUE(org_id,run_id,input_revision,employee_id,resolution),
  UNIQUE(org_id,run_id,input_revision,employee_id,line_id,custody_digest),
  CHECK((resolution='RESOLVED' AND num_nonnulls(gross_won,taxable_monthly_won,pension_standard_monthly_income_won,monthly_standard_hours,income_tax_won,local_tax_won,tax_table_edition)=7 AND jsonb_array_length(blockers)=0)
    OR (resolution='BLOCKED' AND num_nonnulls(gross_won,taxable_monthly_won,pension_standard_monthly_income_won,monthly_standard_hours,income_tax_won,local_tax_won,tax_table_edition)=0 AND jsonb_array_length(blockers)>0)),
  FOREIGN KEY(org_id,run_id,input_revision) REFERENCES public.payroll_input_revisions(org_id,run_id,input_revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,run_id,line_id,employee_id) REFERENCES public.payroll_draft_lines(org_id,run_id,id,employee_id) ON DELETE RESTRICT
);
CREATE TABLE public.payroll_input_source_refs (
  org_id uuid NOT NULL, run_id uuid NOT NULL, input_revision bigint NOT NULL, employee_id uuid NOT NULL,
  source_kind text NOT NULL CHECK(source_kind IN ('EMPLOYMENT_BINDING','EMPLOYMENT_REVISION','WAGE','ATTENDANCE_COVERAGE','ELIGIBILITY','TAX_EVIDENCE')),
  selection text NOT NULL CHECK(selection IN ('SELECTED','CANDIDATE')),
  ordinal smallint NOT NULL CHECK(ordinal BETWEEN 0 AND 255),
  reference_id uuid NOT NULL, reference_revision bigint NOT NULL,
  employment_id uuid, from_us bigint, to_us bigint,
  reference_digest bytea NOT NULL CHECK(octet_length(reference_digest)=32),
  binding_id uuid GENERATED ALWAYS AS (CASE WHEN source_kind='EMPLOYMENT_BINDING' THEN reference_id END) STORED,
  employment_revision_id uuid GENERATED ALWAYS AS (CASE WHEN source_kind='EMPLOYMENT_REVISION' THEN reference_id END) STORED,
  wage_base_id uuid GENERATED ALWAYS AS (CASE WHEN source_kind='WAGE' THEN reference_id END) STORED,
  wage_correction_revision bigint GENERATED ALWAYS AS (CASE WHEN source_kind='WAGE' THEN NULLIF(reference_revision,0) END) STORED,
  coverage_id uuid GENERATED ALWAYS AS (CASE WHEN source_kind='ATTENDANCE_COVERAGE' THEN reference_id END) STORED,
  assessment_id uuid GENERATED ALWAYS AS (CASE WHEN source_kind='ELIGIBILITY' THEN reference_id END) STORED,
  evidence_id uuid GENERATED ALWAYS AS (CASE WHEN source_kind='TAX_EVIDENCE' THEN reference_id END) STORED,
  PRIMARY KEY(org_id,run_id,input_revision,employee_id,source_kind,selection,ordinal),
  CHECK((source_kind='WAGE' AND reference_revision>=0) OR (source_kind<>'WAGE' AND reference_revision>0)),
  CHECK((selection='CANDIDATE' AND ordinal BETWEEN 1 AND 32)
    OR (selection='SELECTED' AND ((source_kind='EMPLOYMENT_REVISION' AND ordinal BETWEEN 1 AND 128) OR (source_kind<>'EMPLOYMENT_REVISION' AND ordinal=0)))),
  CHECK((source_kind='EMPLOYMENT_REVISION' AND employment_id IS NOT NULL AND from_us IS NOT NULL AND to_us IS NOT NULL AND to_us>from_us)
    OR (source_kind='EMPLOYMENT_BINDING' AND employment_id IS NOT NULL AND from_us IS NULL AND to_us IS NULL)
    OR (source_kind NOT IN ('EMPLOYMENT_REVISION','EMPLOYMENT_BINDING') AND employment_id IS NULL AND from_us IS NULL AND to_us IS NULL)),
  FOREIGN KEY(org_id,run_id,input_revision,employee_id) REFERENCES public.payroll_input_units(org_id,run_id,input_revision,employee_id) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,binding_id,reference_revision) REFERENCES public.employment_binding_revisions(org_id,binding_id,revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,employment_id,employment_revision_id,reference_revision) REFERENCES public.employment_revisions(org_id,employment_id,id,version) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,wage_base_id,employee_id) REFERENCES public.employee_contract_wages(org_id,id,employee_id) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,wage_base_id,wage_correction_revision) REFERENCES public.employee_contract_wage_corrections(org_id,base_wage_id,revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,coverage_id,reference_revision) REFERENCES public.attendance_coverage_revisions(org_id,coverage_id,revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,assessment_id,reference_revision) REFERENCES public.payroll_eligibility_assessment_revisions(org_id,assessment_id,revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,evidence_id,reference_revision) REFERENCES public.payroll_tax_evidence_revisions(org_id,evidence_id,revision) ON DELETE RESTRICT
);

CREATE TABLE public.payroll_calculation_batches (
  org_id uuid NOT NULL, run_id uuid NOT NULL, calculation_revision integer NOT NULL CHECK(calculation_revision>0),
  input_revision bigint NOT NULL,
  close_basis_id uuid NOT NULL, close_basis_revision bigint NOT NULL CHECK(close_basis_revision>0),
  member_count integer NOT NULL CHECK(member_count BETWEEN 1 AND 2000),
  success_count integer NOT NULL CHECK(success_count>=0), blocked_count integer NOT NULL CHECK(blocked_count>=0),
  input_manifest_digest bytea NOT NULL CHECK(octet_length(input_manifest_digest)=32),
  header_digest bytea NOT NULL CHECK(octet_length(header_digest)=32),
  batch_digest bytea NOT NULL CHECK(octet_length(batch_digest)=32),
  actor_account_id uuid NOT NULL, command_id uuid NOT NULL, recorded_at_us bigint NOT NULL,
  PRIMARY KEY(org_id,run_id,calculation_revision), UNIQUE(org_id,run_id,calculation_revision,input_revision), UNIQUE(org_id,command_id),
  CHECK(success_count+blocked_count=member_count),
  FOREIGN KEY(org_id,run_id,input_revision) REFERENCES public.payroll_input_revisions(org_id,run_id,input_revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,close_basis_id,close_basis_revision) REFERENCES public.attendance_native_close_bases(org_id,close_basis_id,revision) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,actor_account_id) REFERENCES public.company_actors(org_id,account_id) ON DELETE RESTRICT,
  FOREIGN KEY(org_id,command_id) REFERENCES public.ont_action_command_receipts(org_id,command_id) ON DELETE RESTRICT DEFERRABLE INITIALLY DEFERRED
);
